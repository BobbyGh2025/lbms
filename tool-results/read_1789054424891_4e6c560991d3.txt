   123→function fmt(iso: string | null) {
   124→  if (!iso) return "—";
   125→  try {
   126→    return new Date(iso).toLocaleDateString(undefined, {
   127→      month: "short", day: "2-digit", year: "numeric",
   128→    });
   129→  } catch { return "—"; }
   130→}
   131→
   132→function fmtDateTime(iso: string | null) {
   133→  if (!iso) return "—";
   134→  try {
   135→    return new Date(iso).toLocaleString(undefined, {
   136→      month: "short", day: "2-digit", year: "numeric",
   137→      hour: "2-digit", minute: "2-digit",
   138→    });
   139→  } catch { return "—"; }
   140→}
   141→
   142→function money(v: string | null | undefined): string {
   143→  if (!v) return "GHS 0.00";
   144→  return formatMoney(v, "GHS");
   145→}
   146→
   147→function customerLabel(c: InvoiceDetail["customer"] | null) {
   148→  if (!c) return "—";
   149→  return c.tradingName || c.legalName
   150→    || [c.firstName, c.lastName].filter(Boolean).join(" ")
   151→    || c.customerNumber;
   152→}

   159→    </div>
   160→  );
   161→}

   242→    if (!id) return;

   244→      const res = await fetch(`/api/sales/invoices/${id}`);

   250→    if (!id) return;

   253→      const res = await fetch(`/api/sales/invoices/${id}/${endpoint}`, { method: "POST" });

   255→        const err = await res.json().catch(() => ({}));
   256→        throw new Error(err.error || `Failed to ${endpoint} invoice.`);

   258→      toast.success(successMsg);
   259→      await refresh();
   260→    } catch (err) {
   261→      toast.error(err instanceof Error ? err.message : `Failed to ${endpoint} invoice.`);
   262→    } finally {
   263→      setBusy(false);

   273→      const res = await fetch(`/api/sales/invoices/${id}/items`, {

   277→          description: iDesc.trim(),
   278→          quantity: iQty,
   279→          unitPrice: iPrice,

   282→          inventoryItemId: iItemId || undefined,
   283→        }),
   284→      });
   285→      if (!res.ok) {
   286→        const err = await res.json().catch(() => ({}));
   287→        throw new Error(err.error || "Failed to add item.");
   288→      }
   289→      toast.success("Item added.");

   291→      setIDesc(""); setIQty(""); setIPrice("");
   292→      setIDiscount(""); setITaxRate("0"); setIItemId("");

   295→      toast.error(err instanceof Error ? err.message : "Failed to add item.");
   296→    } finally {
   297→      setSavingItem(false);
   298→    }
   299→  }

   301→  async function handleRecordPayment() {
   302→    if (!data) return;
   303→    if (!pAmount || Number(pAmount) <= 0) {
   304→      toast.error("Amount must be greater than zero."); return;
   305→    }
   306→    // Overpayment guard (client-side; server enforces too).
   307→    if (Number(pAmount) > Number(data.balanceDue)) {
   308→      toast.error(`Amount exceeds balance due (${money(data.balanceDue)}).`);
   309→      return;
   310→    }
   311→    setSavingPay(true);
   312→    try {
   313→      const res = await fetch("/api/sales/payments", {
   314→        method: "POST",
   315→        headers: { "Content-Type": "application/json" },
   316→        body: JSON.stringify({
   317→          customerId: data.customerId,
   318→          invoiceId: data.id,
   319→          amount: pAmount,
   320→          paymentMethod: pMethod,
   321→          reference: pReference.trim() || undefined,
   322→          notes: pNotes.trim() || undefined,
   323→          paymentDate: pDate || undefined,
   324→        }),
   325→      });
   326→      if (!res.ok) {
   327→        const err = await res.json().catch(() => ({}));
   328→        throw new Error(err.error || "Failed to record payment.");
   329→      }
   330→      const created = await res.json();
   331→      toast.success(`Payment recorded: ${created.paymentNumber} (draft — post it to apply).`);
   332→      setPayOpen(false);
   333→      setPAmount(""); setPMethod("cash"); setPReference(""); setPNotes(""); setPDate("");
   334→      await refresh();
   335→    } catch (err) {
   336→      toast.error(err instanceof Error ? err.message : "Failed to record payment.");
   337→    } finally {
   338→      setSavingPay(false);
   339→    }
   340→  }

   353→      toast.error(err instanceof Error ? err.message : "Failed to post payment.");

   355→      setBusy(false);
   356→    }
   357→  }

   371→      toast.error(err instanceof Error ? err.message : "Failed to void payment.");

   373→      setBusy(false);
