   543→// ============================================================================
   544→// PHASE 2 — FINANCE FOUNDATION
   545→// ----------------------------------------------------------------------------
   546→// Journal/ledger double-entry architecture. Money uses Prisma `Decimal`
   547→// (stored as TEXT in SQLite for dev; migrates to DECIMAL(18,2) on
   548→// PostgreSQL/MySQL with zero schema change). Balances are DERIVED from
   549→// posted JournalEntries — there is no mutable `currentBalance` field.
   550→//
   551→//   FinancialAccount  (where money lives: cash/bank/momo)
   552→//          ↓
   553→//   LedgerAccount      (chart of accounts: income/expense/asset categories)
   554→//          ↓
   555→//   Journal            (one financial event; header)
   556→//          ↓
   557→//   JournalEntry       (debit/credit lines; MUST balance: Σ debit = Σ credit)
   558→//          ↓
   559→//   Reporting Service  (single source of truth for balances/summaries)
   560→//          ↓
   561→//   Dashboard / Reports / Exports
   562→// ============================================================================
   563→
   564→// Account types classify the nature of a financial account.
   565→//   asset      — cash, bank, mobile money (debit-normal: debits increase)
   566→//   liability  — payables, loans (credit-normal: credits increase)
   567→//   equity     — owner's equity, retained earnings (credit-normal)
   568→//   income     — revenue categories (credit-normal)
   569→//   expense    — expense categories (debit-normal)
   570→
   571→model FinancialAccount {
   572→  id            String   @id @default(cuid())
   573→  code          String   @unique // e.g. "CASH-001", "BANK-001"
   574→  name          String   @unique
   575→  accountType   String   @default("asset") // asset | liability (financial accounts are typically asset accounts that hold money)
   576→  currency      String   @default("GHS")
   577→  // Opening balance is posted via an OPENING_BALANCE journal — NOT a mutable
   578→  // live field. `openingBalance` is the seed value used to generate that
   579→  // journal at account creation. It is never updated by transactions.
   580→  openingBalance Decimal @default(0)
   581→  status        String   @default("active") // active | inactive
   582→  description   String?
   583→  bankName      String?
   584→  accountNumber String?  // masked/reference only
   585→  createdById   String?
   586→  createdBy     User?    @relation("FinAccountCreatedBy", fields: [createdById], references: [id], onDelete: SetNull)
   587→  createdAt     DateTime @default(now())
   588→  updatedAt     DateTime @updatedAt
   589→  deletedAt     DateTime?
   590→
   591→  journalEntries JournalEntry[]
   592→  journals       Journal[] // journals where this is the primary financial account
   593→
   594→  @@index([status])
   595→  @@index([currency])
   596→}
   597→
   598→// Chart of accounts — categorizes the nature of money movement.
   599→// One LedgerAccount per income/expense/asset/liability/equity category.
   600→model LedgerAccount {
   601→  id          String   @id @default(cuid())
   602→  code        String   @unique // e.g. "INC-SALES", "EXP-FUEL"
   603→  name        String   @unique

   624→// A Journal is ONE financial event (an income receipt, an expense payment,
   625→// a transfer, an opening balance, or an adjustment). It owns ≥2 JournalEntry
   626→// lines that must balance: Σ(debit) = Σ(credit).
   627→model Journal {
   628→  id              String   @id @default(cuid())
   629→  reference       String   @unique // e.g. "INC-2026-000001" — concurrency-safe, generated in-transaction
   630→  transactionType String   // income | expense | transfer | opening_balance | adjustment
   631→  status          String   @default("posted") // draft | posted | voided | reversed
   632→  transactionDate DateTime // the accounting date (NOT createdAt — may differ)
   633→  description     String?
   634→  notes           String?
   635→
   636→  // The financial account primarily affected (receiving account for income,
   637→  // paying account for expense, source for transfer). For multi-account
   638→  // journals (transfers), the full debit/credit breakdown is in JournalEntry.
   639→  financialAccountId String?
   640→  financialAccount   FinancialAccount? @relation(fields: [financialAccountId], references: [id], onDelete: Restrict)
   641→
   642→  // The primary ledger category (income/expense category). Full breakdown
   643→  // is in JournalEntry.
   644→  ledgerAccountId String?
   645→  ledgerAccount   LedgerAccount? @relation(fields: [ledgerAccountId], references: [id], onDelete: Restrict)
   646→
   647→  // Optional relations — wired to Phase 5/6 entities when they land.
   648→  departmentId   String?
   649→  department     Department? @relation(fields: [departmentId], references: [id], onDelete: SetNull)
   650→
   651→  // Party reference (customer/supplier) — nullable stub from Phase 2.
   652→  // `partyType` discriminates; `partyRef` holds the entity ID as a string.
   653→  // Phase 4 adds proper FK relations (customerId/supplierId) below.
   654→  partyType String? // customer | supplier | null
   655→  partyRef  String? // future FK to Customer/Supplier
   656→
   657→  // Phase 4: proper Customer/Supplier FK relations (nullable, alongside the
   658→  // string stubs for backward compatibility with existing journals).
   659→  customerId String?
   660→  customer   Customer? @relation("JournalCustomer", fields: [customerId], references: [id], onDelete: SetNull)
   661→  supplierId String?
   662→  supplier   Supplier? @relation("JournalSupplier", fields: [supplierId], references: [id], onDelete: SetNull)
   663→
   664→  // Project reference — nullable stub for Phase 6.
   665→  projectRef String?
   666→
   667→  paymentMethod String? // cash | bank_transfer | mobile_money | card | cheque | other
   668→  externalRef  String? // bank slip / cheque / momo transaction id
   669→
   670→  // Reversal linkage. `reversesId` is the FK on the reversal journal pointing
   671→  // to the original. `reverses` is the relation (holds fields). `reversedBy`
   672→  // is the inverse relation on the original journal (no fields).
   673→  reversesId      String?  @unique
   674→  reverses        Journal?  @relation("JournalReversal", fields: [reversesId], references: [id], onDelete: Restrict)
   675→  reversedBy     Journal?  @relation("JournalReversal")
   676→  reversalReason String?
   677→
   678→  // Total monetary value of the transaction (the absolute amount; equals
   679→  // Σ(debit) = Σ(credit)). Stored as Decimal for exact precision.
   680→  amount      Decimal
   681→  currency    String  @default("GHS")
   682→
   683→  createdById String
   684→  createdBy   User    @relation("JournalCreatedBy", fields: [createdById], references: [id], onDelete: Restrict)
   685→  postedAt    DateTime?
   686→  voidedAt    DateTime?
   687→  voidedById  String?
   688→  voidedBy    User?   @relation("JournalVoidedBy", fields: [voidedById], references: [id], onDelete: SetNull)
   689→  createdAt   DateTime @default(now())
   690→  updatedAt   DateTime @updatedAt
   691→
   692→  entries JournalEntry[]
   693→
   694→  @@index([transactionType])
   695→  @@index([status])
   696→  @@index([transactionDate])
   697→  @@index([financialAccountId])
   698→  @@index([ledgerAccountId])
   699→  @@index([departmentId])
   700→  @@index([createdById])
   701→  @@index([reference])
   702→  @@index([customerId])
   703→  @@index([supplierId])
   704→}

   706→// A single debit OR credit line within a Journal. Exactly one of
   707→// (debit, credit) is non-zero per entry. The journal is balanced when
   708→// Σ(debit) = Σ(credit) across all its entries.
   709→//
   710→// `financialAccountId` is OPTIONAL: only entries that move money in/out of
   711→// a cash/bank/momo account reference a FinancialAccount. The counter-side
   712→// (e.g. an income category credit, or an expense category debit) references
   713→// only a LedgerAccount. This mirrors real accounting where income/expense
   714→// categories are not "accounts you hold money in".
   715→model JournalEntry {
   716→  id                String   @id @default(cuid())
   717→  journalId         String
   718→  journal           Journal  @relation(fields: [journalId], references: [id], onDelete: Cascade)
   719→
   720→  financialAccountId String?
   721→  financialAccount   FinancialAccount? @relation(fields: [financialAccountId], references: [id], onDelete: Restrict)
   722→
   723→  ledgerAccountId   String?
   724→  ledgerAccount     LedgerAccount? @relation(fields: [ledgerAccountId], references: [id], onDelete: Restrict)
   725→
   726→  // Exactly one of debit/credit is non-zero (the other is 0). Both Decimal.
   727→  debit  Decimal @default(0)
   728→  credit Decimal @default(0)
   729→  currency String @default("GHS")
   730→
   731→  description String?
   732→  createdAt   DateTime @default(now())
   733→
   734→  @@index([journalId])
   735→  @@index([financialAccountId])
   736→  @@index([ledgerAccountId])
   737→}

   742→model FinanceRefCounter {
