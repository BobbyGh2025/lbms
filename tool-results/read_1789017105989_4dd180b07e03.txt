   125→function isOverdue(dateStr: string | null, status: string, terminal: Set<string>) {
   126→  if (!dateStr || terminal.has(status)) return false;
   127→  try {
   128→    return new Date(dateStr) < new Date();
   129→  } catch { return false; }
   130→}
   131→
   132→function fmt(iso: string | null, withYear = false) {
   133→  if (!iso) return "—";
   134→  try {
   135→    return new Date(iso).toLocaleDateString(undefined, withYear
   136→      ? { month: "short", day: "2-digit", year: "numeric" }
   137→      : { month: "short", day: "2-digit" });
   138→  } catch { return "—"; }
   139→}
   140→
   141→function supplierLabel(s: { supplierNumber: string; tradingName: string | null; legalName: string | null } | null | undefined) {
   142→  if (!s) return "—";
   143→  return s.tradingName || s.legalName || s.supplierNumber;
   144→}
