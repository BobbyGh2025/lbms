   162→function formatDateTime(iso: string | null) {
   163→  if (!iso) return "—";
   164→  const d = new Date(iso);
   165→  if (Number.isNaN(d.getTime())) return "—";
   166→  return d.toLocaleString(undefined, {
   167→    year: "numeric",
   168→    month: "short",
   169→    day: "2-digit",
   170→    hour: "2-digit",
   171→    minute: "2-digit",
   172→  });
   173→}
   174→
   175→function formatDate(iso: string | null) {
   176→  if (!iso) return "—";
   177→  const d = new Date(iso);
   178→  if (Number.isNaN(d.getTime())) return "—";
   179→  return d.toLocaleDateString(undefined, {
   180→    year: "numeric",
   181→    month: "short",
   182→    day: "2-digit",
   183→  });
   184→}
   185→
   186→function initials(name: string) {
   187→  const parts = name.trim().split(/\s+/).filter(Boolean);
   188→  if (parts.length === 0) return "?";
   189→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   190→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   191→}
   192→
   193→async function readError(res: Response): Promise<string> {
   194→  try {
   195→    const data = await res.json();
   196→    if (data?.error) return data.error as string;
   197→  } catch {
   198→    // ignore — fall through to status text
   199→  }
   200→  if (res.status === 401) return "You are not signed in.";
   201→  if (res.status === 403) return "You are not authorized to perform this action.";
   202→  if (res.status === 404) return "The user was not found.";
   203→  if (res.status === 400) return "The request was invalid.";
   204→  return `Request failed (${res.status}).`;
   205→}

   514→      />

   519→          refresh();
   520→        }}
   521→      />
   522→      <ResetPasswordDialog
   523→        user={resetUser}
   524→        onOpenChange={(o) => {
   525→          if (!o) setResetUser(null);

   527→        }}
   528→      />
   529→      <ConfirmDialog
   530→        open={!!deleteUser}
   531→        onOpenChange={(o) => {
   532→          if (!o) setDeleteUser(null);
   533→        }}
   534→        trigger={null}
   535→        title="Delete user account?"
   536→        description={
   537→          deleteUser
   538→            ? `This will soft-delete ${deleteUser.username} (${deleteUser.email}). The user will no longer be able to sign in, but the record is retained for audit history.`
   539→            : ""
   540→        }
   541→        confirmLabel="Delete"
   542→        destructive
   543→        onConfirm={async () => {
   544→          if (!deleteUser) return;
   545→          const u = deleteUser;
   546→          try {
   547→            const res = await fetch(`/api/users/${u.id}`, { method: "DELETE" });
   548→            if (!res.ok) throw new Error(await readError(res));
   549→            toast.success(`User ${u.username} deleted.`);
   550→            refresh();
   551→          } catch (err) {
   552→            toast.error(err instanceof Error ? err.message : "Failed to delete user.");
   553→            // Re-throw so ConfirmDialog keeps the dialog open for retry.
   554→            throw err;
   555→          }
   556→        }}
   557→      />
   558→    </div>
   559→  );
   560→}
