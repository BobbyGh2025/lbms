   123→   123→function fmt(iso: string | null) {
   124→   124→  if (!iso) return "—";
   125→   125→  try {
   126→   126→    return new Date(iso).toLocaleDateString(undefined, {
   127→   127→      month: "short", day: "2-digit", year: "numeric",
   128→   128→    });
   129→   129→  } catch { return "—"; }
   130→   130→}
   131→   131→
   132→   132→function fmtDateTime(iso: string | null) {
   133→   133→  if (!iso) return "—";
   134→   134→  try {
   135→   135→    return new Date(iso).toLocaleString(undefined, {
   136→   136→      month: "short", day: "2-digit", year: "numeric",
   137→   137→      hour: "2-digit", minute: "2-digit",
   138→   138→    });
   139→   139→  } catch { return "—"; }
   140→   140→}
   141→   141→
   142→   142→function money(v: string | null | undefined): string {
   143→   143→  if (!v) return "GHS 0.00";
   144→   144→  return formatMoney(v, "GHS");
   145→   145→}
