   161→   161→function customerLabel(c: QuoteListItem["customer"]) {
   162→   162→  if (!c) return "—";
   163→   163→  return c.tradingName || c.legalName
   164→   164→    || [c.firstName, c.lastName].filter(Boolean).join(" ")
   165→   165→    || c.customerNumber || "—";
   166→   166→}
   167→   167→
   168→   168→function fmt(iso: string | null) {
   169→   169→  if (!iso) return "—";
   170→   170→  try {
   171→   171→    return new Date(iso).toLocaleDateString(undefined, {
   172→   172→      month: "short", day: "2-digit", year: "numeric",
   173→   173→    });
   174→   174→  } catch { return "—"; }
   175→   175→}
   176→   176→
   177→   177→function money(v: string | null | undefined): string {
   178→   178→  if (!v) return "GHS 0.00";
   179→   179→  return formatMoney(v, "GHS");
   180→   180→}
