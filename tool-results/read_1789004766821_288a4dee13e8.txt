   125→   125→function isOverdue(dateStr: string | null, status: string, terminal: Set<string>) {
   126→   126→  if (!dateStr || terminal.has(status)) return false;
   127→   127→  try {
   128→   128→    return new Date(dateStr) < new Date();
   129→   129→  } catch { return false; }
   130→   130→}
   131→   131→
   132→   132→function fmt(iso: string | null, withYear = false) {
   133→   133→  if (!iso) return "—";
   134→   134→  try {
   135→   135→    return new Date(iso).toLocaleDateString(undefined, withYear
   136→   136→      ? { month: "short", day: "2-digit", year: "numeric" }
   137→   137→      : { month: "short", day: "2-digit" });
   138→   138→  } catch { return "—"; }
   139→   139→}
   140→   140→
   141→   141→function supplierLabel(s: { supplierNumber: string; tradingName: string | null; legalName: string | null } | null | undefined) {
   142→   142→  if (!s) return "—";
   143→   143→  return s.tradingName || s.legalName || s.supplierNumber;
   144→   144→}
