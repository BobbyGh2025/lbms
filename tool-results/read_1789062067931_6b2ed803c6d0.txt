   195→function supplierLabel(s: SupplierLite | null): string {
   196→  if (!s) return "—";
   197→  return s.tradingName || s.legalName || s.supplierNumber;
   198→}
   199→
   200→function fmt(iso: string | null) {
   201→  if (!iso) return "—";
   202→  try {
   203→    return new Date(iso).toLocaleDateString(undefined, {
   204→      month: "short", day: "2-digit", year: "numeric",
   205→    });
   206→  } catch { return "—"; }
   207→}
   208→
   209→function fmtDateTime(iso: string | null) {
   210→  if (!iso) return "—";
   211→  try {
   212→    return new Date(iso).toLocaleString(undefined, {
   213→      month: "short", day: "2-digit", year: "numeric",
   214→      hour: "2-digit", minute: "2-digit",
   215→    });
   216→  } catch { return "—"; }
   217→}
   218→
   219→function money(v: string | null | undefined): string {
   220→  if (!v) return "GHS 0.00";
   221→  return formatMoney(v, "GHS");
   222→}

   502→  async function handleCreatePayment() {
   503→    if (!pSupplierId) { toast.error("Supplier is required."); return; }
   504→    if (!pFinAccountId) { toast.error("Paying financial account is required."); return; }
   505→    if (!pAmount || Number(pAmount) <= 0) {
   506→      toast.error("Amount must be greater than zero."); return;
   507→    }
   508→    setSavingPayment(true);
   509→    try {
   510→      const res = await fetch("/api/payables/payments", {
   511→        method: "POST",
   512→        headers: { "Content-Type": "application/json" },
   513→        body: JSON.stringify({
   514→          supplierId: pSupplierId,
   515→          supplierBillId: pBillId || undefined,
   516→          financialAccountId: pFinAccountId,
   517→          amount: pAmount,
   518→          paymentMethod: pMethod,
   519→          reference: pReference.trim() || undefined,
   520→          notes: pNotes.trim() || undefined,
   521→        }),
   522→      });
   523→      if (!res.ok) {
   524→        const err = await res.json().catch(() => ({}));
   525→        throw new Error(err.error || "Failed to record payment.");
