    99→function formatDate(iso: string | null) {
   100→  if (!iso) return "—";
   101→  const d = new Date(iso);
   102→  if (Number.isNaN(d.getTime())) return "—";
   103→  return d.toLocaleDateString(undefined, { year: "numeric", month: "short", day: "2-digit" });
   104→}
   105→
   106→function todayISO() {
   107→  // Local YYYY-MM-DD (the API accepts date strings).
   108→  const d = new Date();
   109→  const y = d.getFullYear();
   110→  const m = String(d.getMonth() + 1).padStart(2, "0");
   111→  const day = String(d.getDate()).padStart(2, "0");
   112→  return `${y}-${m}-${day}`;
   113→}

   131→async function readError(res: Response): Promise<string> {
   132→  try {
   133→    const data = await res.json();
   134→    if (data?.error) return data.error as string;
   135→    if (data?.details) return JSON.stringify(data.details);
   136→  } catch {
   137→    /* noop */
   138→  }
   139→  if (res.status === 401) return "You are not signed in.";
   140→  if (res.status === 403) return "You are not authorized to perform this action.";
   141→  return `Request failed (${res.status}).`;
   142→}

   442→function RecordIncomeDialog({ open, onOpenChange }: RecordIncomeDialogProps) {
   443→  const [date, setDate] = useState(todayISO());
   444→  const [amount, setAmount] = useState("");
   445→  const [accountId, setAccountId] = useState("");
   446→  const [categoryId, setCategoryId] = useState("");
   447→  const [description, setDescription] = useState("");
   448→  const [paymentMethod, setPaymentMethod] = useState("");
   449→  const [externalRef, setExternalRef] = useState("");
   450→  const [departmentId, setDepartmentId] = useState("__none__");
   451→  const [notes, setNotes] = useState("");
   452→  const [status, setStatus] = useState<"posted" | "draft">("posted");
   453→  const [saving, setSaving] = useState(false);
   454→
   455→  const [accounts, setAccounts] = useState<AccountOption[]>([]);
   456→  const [categories, setCategories] = useState<CategoryOption[]>([]);
   457→  const [departments, setDepartments] = useState<DepartmentOption[]>([]);
   458→
   459→  // Reset form on open
   460→  useEffect(() => {
   461→    if (!open) return;
   462→    setDate(todayISO());
   463→    setAmount("");
   464→    setAccountId("");
   465→    setCategoryId("");
   466→    setDescription("");
   467→    setPaymentMethod("");
   468→    setExternalRef("");
   469→    setDepartmentId("__none__");
   470→    setNotes("");
   471→    setStatus("posted");
   472→  }, [open]);
   473→
   474→  // Fetch supporting options when dialog opens
   475→  useEffect(() => {
   476→    if (!open) return;
   477→    let cancelled = false;
   478→    (async () => {
   479→      try {
   480→        const [accRes, catRes, depRes] = await Promise.all([
   481→          fetch("/api/finance/accounts"),
   482→          fetch("/api/finance/categories?accountClass=income"),
   483→          fetch("/api/departments"),
   484→        ]);
   485→        const [acc, cat, dep] = await Promise.all([
   486→          accRes.ok ? accRes.json() : { items: [] },
   487→          catRes.ok ? catRes.json() : { items: [] },
   488→          depRes.ok ? depRes.json() : { items: [] },
   489→        ]);
   490→        if (cancelled) return;
   491→        setAccounts(acc.items ?? []);
   492→        setCategories(cat.items ?? []);
   493→        setDepartments(dep.items ?? []);
   494→      } catch {
   495→        if (!cancelled) {
   496→          setAccounts([]);
   497→          setCategories([]);
   498→          setDepartments([]);
   499→        }
   500→      }
   501→    })();
   502→    return () => {
   503→      cancelled = true;
   504→    };
   505→  }, [open]);
   506→
   507→  const selectedAccount = accounts.find((a) => a.id === accountId);
   508→  const currency = selectedAccount?.currency ?? "GHS";
   509→
   510→  const valid =
   511→    !!date &&
   512→    !!amount &&
   513→    Number(amount) > 0 &&
   514→    !!accountId &&
   515→    !!categoryId;
   516→
   517→  async function handleSubmit() {
   518→    if (!valid) {
   519→      toast.error("Please fill in date, amount, account, and category.");
   520→      return;
   521→    }
   522→    setSaving(true);
   523→    try {
   524→      const payload: Record<string, unknown> = {
   525→        date,
   526→        amount: String(amount),
   527→        financialAccountId: accountId,
   528→        ledgerAccountId: categoryId,
   529→        description: description.trim() || undefined,
   530→        notes: notes.trim() || undefined,
   531→        paymentMethod: paymentMethod || undefined,
   532→        externalRef: externalRef.trim() || undefined,
   533→        status,
   534→      };
   535→      if (departmentId !== "__none__") payload.departmentId = departmentId;
   536→      const res = await fetch("/api/finance/income", {
   537→        method: "POST",
   538→        headers: { "Content-Type": "application/json" },
   539→        body: JSON.stringify(payload),
   540→      });
   541→      if (!res.ok) throw new Error(await readError(res));
   542→      toast.success("Income recorded.");
   543→      onOpenChange(false);
   544→    } catch (err) {
   545→      toast.error(err instanceof Error ? err.message : "Failed to record income.");
   546→    } finally {
   547→      setSaving(false);
   548→    }
   549→  }
   550→
   551→  return (
   552→    <Dialog open={open} onOpenChange={onOpenChange}>
   553→      <DialogContent className="sm:max-w-[560px]">
   554→        <DialogHeader>
   555→          <DialogTitle>Record Income</DialogTitle>
   556→          <DialogDescription>
   557→            Post an income transaction. Money is debited to the selected cash/bank account and credited to the chosen income category.
   558→          </DialogDescription>
   559→        </DialogHeader>
   560→
   561→        <div className="grid gap-4">
   562→          <div className="grid grid-cols-2 gap-3">
   563→            <div className="space-y-1.5">
   564→              <Label htmlFor="inc-date">Date</Label>
   565→              <Input
   566→                id="inc-date"
   567→                type="date"
   568→                value={date}
   569→                onChange={(e) => setDate(e.target.value)}
   570→              />
   571→            </div>
   572→            <div className="space-y-1.5">
   573→              <Label htmlFor="inc-amount">
   574→                Amount {currency ? <span className="text-muted-foreground">({currency})</span> : null}
   575→              </Label>
   576→              <Input
   577→                id="inc-amount"
   578→                type="number"
   579→                step="0.01"
   580→                min="0"
   581→                inputMode="decimal"
   582→                placeholder="0.00"
   583→                value={amount}
   584→                onChange={(e) => setAmount(e.target.value)}
   585→              />
   586→            </div>
   587→          </div>
   588→
   589→          <div className="space-y-1.5">
   590→            <Label htmlFor="inc-account">Cash / Bank Account</Label>
   591→            <Select value={accountId} onValueChange={setAccountId}>
   592→              <SelectTrigger id="inc-account" className="w-full">
   593→                <SelectValue placeholder="Select account…" />
   594→              </SelectTrigger>
   595→              <SelectContent>
   596→                {accounts.length === 0 ? (
   597→                  <SelectItem value="__none__" disabled>
   598→                    No active accounts
   599→                  </SelectItem>
   600→                ) : (
   601→                  accounts.map((a) => (
   602→                    <SelectItem key={a.id} value={a.id}>
   603→                      {a.code} · {a.name} ({a.currency})
   604→                    </SelectItem>
   605→                  ))
   606→                )}
   607→              </SelectContent>
   608→            </Select>
   609→          </div>
   610→
   611→          <div className="space-y-1.5">
   612→            <Label htmlFor="inc-category">Income Category</Label>
   613→            <Select value={categoryId} onValueChange={setCategoryId}>
