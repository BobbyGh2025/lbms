   162→   162→function formatDateTime(iso: string | null) {
   163→   163→  if (!iso) return "—";
   164→   164→  const d = new Date(iso);
   165→   165→  if (Number.isNaN(d.getTime())) return "—";
   166→   166→  return d.toLocaleString(undefined, {
   167→   167→    year: "numeric",
   168→   168→    month: "short",
   169→   169→    day: "2-digit",
   170→   170→    hour: "2-digit",
   171→   171→    minute: "2-digit",
   172→   172→  });
   173→   173→}
   174→   174→
   175→   175→function formatDate(iso: string | null) {
   176→   176→  if (!iso) return "—";
   177→   177→  const d = new Date(iso);
   178→   178→  if (Number.isNaN(d.getTime())) return "—";
   179→   179→  return d.toLocaleDateString(undefined, {
   180→   180→    year: "numeric",
   181→   181→    month: "short",
   182→   182→    day: "2-digit",
   183→   183→  });
   184→   184→}
   185→   185→
   186→   186→function initials(name: string) {
   187→   187→  const parts = name.trim().split(/\s+/).filter(Boolean);
   188→   188→  if (parts.length === 0) return "?";
   189→   189→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   190→   190→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   191→   191→}
   192→   192→
   193→   193→async function readError(res: Response): Promise<string> {
   194→   194→  try {
   195→   195→    const data = await res.json();
   196→   196→    if (data?.error) return data.error as string;
   197→   197→  } catch {
   198→   198→    // ignore — fall through to status text
   199→   199→  }
   200→   200→  if (res.status === 401) return "You are not signed in.";
   201→   201→  if (res.status === 403) return "You are not authorized to perform this action.";
   202→   202→  if (res.status === 404) return "The user was not found.";
   203→   203→  if (res.status === 400) return "The request was invalid.";
   204→   204→  return `Request failed (${res.status}).`;
   205→   205→}

   737→   737→function UserFormDialog({ open, onOpenChange, mode, user }: UserFormDialogProps) {
   738→   738→  const [email, setEmail] = useState("");
   739→   739→  const [username, setUsername] = useState("");
   740→   740→  const [password, setPassword] = useState("");
   741→   741→  const [status, setStatus] = useState<UserStatus>("active");
   742→   742→  const [employeeId, setEmployeeId] = useState<string>("__none__");
   743→   743→  const [selectedRoles, setSelectedRoles] = useState<string[]>([]);
   744→   744→  const [saving, setSaving] = useState(false);
   745→   745→
   746→   746→  const [employees, setEmployees] = useState<EmployeeOption[]>([]);
   747→   747→  const [roles, setRoles] = useState<RoleOption[]>([]);
   748→   748→
   749→   749→  // Pre-fill form when editing
   750→   750→  useEffect(() => {
   751→   751→    if (!open) return;
   752→   752→    if (mode === "edit" && user) {
   753→   753→      setEmail(user.email);
   754→   754→      setUsername(user.username);
   755→   755→      setPassword("");
   756→   756→      setStatus(user.status);
   757→   757→      setEmployeeId(user.employee?.id ?? "__none__");
   758→   758→      setSelectedRoles(user.roles.map((r) => r.id));
   759→   759→    } else {
   760→   760→      setEmail("");
   761→   761→      setUsername("");
   762→   762→      setPassword("");
   763→   763→      setStatus("active");
   764→   764→      setEmployeeId("__none__");
   765→   765→      setSelectedRoles([]);
   766→   766→    }
   767→   767→  }, [open, mode, user]);
   768→   768→
   769→   769→  // Fetch employees (when dialog opens) — include the currently linked one
   770→   770→  // for edit mode so it shows up in the dropdown.
   771→   771→  useEffect(() => {
   772→   772→    if (!open) return;
   773→   773→    let cancelled = false;
   774→   774→    (async () => {
   775→   775→      try {
   776→   776→        const params = new URLSearchParams();
   777→   777→        if (mode === "edit" && user?.employee?.id) {
   778→   778→          params.set("includeEmployeeId", user.employee.id);
   779→   779→        }
   780→   780→        const res = await fetch(`/api/users/employees?${params.toString()}`);
   781→   781→        if (!res.ok) throw new Error(await readError(res));
   782→   782→        const data = await res.json();
   783→   783→        if (!cancelled) setEmployees(data.items ?? []);
   784→   784→      } catch {
   785→   785→        if (!cancelled) setEmployees([]);

   788→   788→    return () => {
   789→   789→      cancelled = true;
   790→   790→    };
   791→   791→  }, [open, mode, user]);
   792→   792→

   794→   794→  useEffect(() => {
   795→   795→    if (!open) return;
   796→   796→    let cancelled = false;

   798→   798→      try {
   799→   799→        const res = await fetch(`/api/users/roles`);

   802→   802→        if (!cancelled) setRoles(data.items ?? []);
   803→   803→      } catch {
   804→   804→        if (!cancelled) setRoles([]);
   805→   805→      }
   806→   806→    })();
   807→   807→    return () => {
   808→   808→      cancelled = true;
   809→   809→    };
   810→   810→  }, [open]);
   811→   811→
   812→   812→  const isEdit = mode === "edit";
   813→   813→  const valid =
   814→   814→    email.trim().length > 0 &&
   815→   815→    /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email.trim()) &&
   816→   816→    username.trim().length >= 3 &&
   817→   817→    (!isEdit ? password.length >= 8 : password.length === 0 || password.length >= 8);
   818→   818→

   820→   820→    e.preventDefault();

   824→   824→      const payload: Record<string, unknown> = {
   825→   825→        email: email.trim().toLowerCase(),
   826→   826→        username: username.trim(),
   827→   827→        roleIds: selectedRoles,

   829→   829→        employeeId: employeeId === "__none__" ? null : employeeId,
   830→   830→      };
   831→   831→      if (!isEdit || password.length > 0) {
   832→   832→        payload.password = password;
   833→   833→      }

   838→   838→        method,
