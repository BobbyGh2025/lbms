   585→// Concurrency-safe task number counter (separate from all other counters).
   586→model TaskRefCounter {
   587→  id        String   @id @default(cuid())
   588→  prefix    String   @default("TSK")
   589→  year      Int
   590→  nextNumber Int     @default(1)
   591→
   592→  @@unique([prefix, year])

   595→model Task {
   596→  id              String   @id @default(cuid())
   597→  taskNumber      String   @unique // TSK-2026-000001
   598→  title           String
   599→  description     String?
   600→  projectId       String?
   601→  project         Project? @relation("ProjectTasks", fields: [projectId], references: [id], onDelete: SetNull)
   602→  customerId      String?  // optional — may be derived from project
   603→  customer        Customer? @relation("CustomerTasks", fields: [customerId], references: [id], onDelete: SetNull)

   733→// Journal/ledger double-entry architecture. Money uses Prisma `Decimal`
   734→// (stored as TEXT in SQLite for dev; migrates to DECIMAL(18,2) on
   735→// PostgreSQL/MySQL with zero schema change). Balances are DERIVED from
   736→// posted JournalEntries — there is no mutable `currentBalance` field.
   737→//
   738→//   FinancialAccount  (where money lives: cash/bank/momo)
   739→//          ↓
   740→//   LedgerAccount      (chart of accounts: income/expense/asset categories)
   741→//          ↓
   742→//   Journal            (one financial event; header)
   743→//          ↓
   744→//   JournalEntry       (debit/credit lines; MUST balance: Σ debit = Σ credit)
   745→//          ↓
   746→//   Reporting Service  (single source of truth for balances/summaries)
   747→//          ↓
   748→//   Dashboard / Reports / Exports
   749→// ============================================================================
   750→
   751→// Account types classify the nature of a financial account.
   752→//   asset      — cash, bank, mobile money (debit-normal: debits increase)
   753→//   liability  — payables, loans (credit-normal: credits increase)
   754→//   equity     — owner's equity, retained earnings (credit-normal)
   755→//   income     — revenue categories (credit-normal)
   756→//   expense    — expense categories (debit-normal)
   757→
   758→model FinancialAccount {
   759→  id            String   @id @default(cuid())
   760→  code          String   @unique // e.g. "CASH-001", "BANK-001"
   761→  name          String   @unique
   762→  accountType   String   @default("asset") // asset | liability (financial accounts are typically asset accounts that hold money)
   763→  currency      String   @default("GHS")
   764→  // Opening balance is posted via an OPENING_BALANCE journal — NOT a mutable
   765→  // live field. `openingBalance` is the seed value used to generate that
   766→  // journal at account creation. It is never updated by transactions.
   767→  openingBalance Decimal @default(0)
   768→  status        String   @default("active") // active | inactive
   769→  description   String?
   770→  bankName      String?
   771→  accountNumber String?  // masked/reference only
   772→  createdById   String?
   773→  createdBy     User?    @relation("FinAccountCreatedBy", fields: [createdById], references: [id], onDelete: SetNull)
   774→  createdAt     DateTime @default(now())
   775→  updatedAt     DateTime @updatedAt
   776→  deletedAt     DateTime?
   777→
   778→  journalEntries JournalEntry[]
   779→  journals       Journal[] // journals where this is the primary financial account
   780→
   781→  @@index([status])
   782→  @@index([currency])
   783→}

   811→// A Journal is ONE financial event (an income receipt, an expense payment,
   812→// a transfer, an opening balance, or an adjustment). It owns ≥2 JournalEntry
   813→// lines that must balance: Σ(debit) = Σ(credit).
   814→model Journal {
   815→  id              String   @id @default(cuid())
   816→  reference       String   @unique // e.g. "INC-2026-000001" — concurrency-safe, generated in-transaction
   817→  transactionType String   // income | expense | transfer | opening_balance | adjustment
   818→  status          String   @default("posted") // draft | posted | voided | reversed
   819→  transactionDate DateTime // the accounting date (NOT createdAt — may differ)
   820→  description     String?
   821→  notes           String?
   822→
   823→  // The financial account primarily affected (receiving account for income,
   824→  // paying account for expense, source for transfer). For multi-account
   825→  // journals (transfers), the full debit/credit breakdown is in JournalEntry.
   826→  financialAccountId String?
   827→  financialAccount   FinancialAccount? @relation(fields: [financialAccountId], references: [id], onDelete: Restrict)
   828→
   829→  // The primary ledger category (income/expense category). Full breakdown
   830→  // is in JournalEntry.
   831→  ledgerAccountId String?

   899→// A single debit OR credit line within a Journal. Exactly one of
   900→// (debit, credit) is non-zero per entry. The journal is balanced when
   901→// Σ(debit) = Σ(credit) across all its entries.
   902→//
   903→// `financialAccountId` is OPTIONAL: only entries that move money in/out of
   904→// a cash/bank/momo account reference a FinancialAccount. The counter-side
   905→// (e.g. an income category credit, or an expense category debit) references
   906→// only a LedgerAccount. This mirrors real accounting where income/expense
   907→// categories are not "accounts you hold money in".
   908→model JournalEntry {
   909→  id                String   @id @default(cuid())
   910→  journalId         String
   911→  journal           Journal  @relation(fields: [journalId], references: [id], onDelete: Cascade)
   912→
   913→  financialAccountId String?
   914→  financialAccount   FinancialAccount? @relation(fields: [financialAccountId], references: [id], onDelete: Restrict)
   915→
   916→  ledgerAccountId   String?
   917→  ledgerAccount     LedgerAccount? @relation(fields: [ledgerAccountId], references: [id], onDelete: Restrict)
   918→
   919→  // Exactly one of debit/credit is non-zero (the other is 0). Both Decimal.
   920→  debit  Decimal @default(0)
   921→  credit Decimal @default(0)
   922→  currency String @default("GHS")
   923→
   924→  description String?
   925→  createdAt   DateTime @default(now())
   926→
   927→  @@index([journalId])
   928→  @@index([financialAccountId])
   929→  @@index([ledgerAccountId])
   930→}
