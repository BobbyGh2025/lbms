   642→// Journal/ledger double-entry architecture. Money uses Prisma `Decimal`
   643→// (stored as TEXT in SQLite for dev; migrates to DECIMAL(18,2) on
   644→// PostgreSQL/MySQL with zero schema change). Balances are DERIVED from
   645→// posted JournalEntries — there is no mutable `currentBalance` field.
   646→//
   647→//   FinancialAccount  (where money lives: cash/bank/momo)
   648→//          ↓
   649→//   LedgerAccount      (chart of accounts: income/expense/asset categories)
   650→//          ↓
   651→//   Journal            (one financial event; header)
   652→//          ↓
   653→//   JournalEntry       (debit/credit lines; MUST balance: Σ debit = Σ credit)
   654→//          ↓
   655→//   Reporting Service  (single source of truth for balances/summaries)
   656→//          ↓
   657→//   Dashboard / Reports / Exports
   658→// ============================================================================
   659→
   660→// Account types classify the nature of a financial account.
   661→//   asset      — cash, bank, mobile money (debit-normal: debits increase)
   662→//   liability  — payables, loans (credit-normal: credits increase)
   663→//   equity     — owner's equity, retained earnings (credit-normal)
   664→//   income     — revenue categories (credit-normal)
   665→//   expense    — expense categories (debit-normal)
   666→
   667→model FinancialAccount {
   668→  id            String   @id @default(cuid())
   669→  code          String   @unique // e.g. "CASH-001", "BANK-001"
   670→  name          String   @unique
   671→  accountType   String   @default("asset") // asset | liability (financial accounts are typically asset accounts that hold money)
   672→  currency      String   @default("GHS")
   673→  // Opening balance is posted via an OPENING_BALANCE journal — NOT a mutable
   674→  // live field. `openingBalance` is the seed value used to generate that
   675→  // journal at account creation. It is never updated by transactions.
   676→  openingBalance Decimal @default(0)
   677→  status        String   @default("active") // active | inactive
   678→  description   String?
   679→  bankName      String?
   680→  accountNumber String?  // masked/reference only
   681→  createdById   String?
   682→  createdBy     User?    @relation("FinAccountCreatedBy", fields: [createdById], references: [id], onDelete: SetNull)
   683→  createdAt     DateTime @default(now())
   684→  updatedAt     DateTime @updatedAt
   685→  deletedAt     DateTime?
   686→
   687→  journalEntries JournalEntry[]
   688→  journals       Journal[] // journals where this is the primary financial account
   689→
   690→  @@index([status])
   691→  @@index([currency])
   692→}

   720→// A Journal is ONE financial event (an income receipt, an expense payment,
   721→// a transfer, an opening balance, or an adjustment). It owns ≥2 JournalEntry
   722→// lines that must balance: Σ(debit) = Σ(credit).
   723→model Journal {
   724→  id              String   @id @default(cuid())
   725→  reference       String   @unique // e.g. "INC-2026-000001" — concurrency-safe, generated in-transaction
   726→  transactionType String   // income | expense | transfer | opening_balance | adjustment
   727→  status          String   @default("posted") // draft | posted | voided | reversed
   728→  transactionDate DateTime // the accounting date (NOT createdAt — may differ)
   729→  description     String?
   730→  notes           String?
   731→
   732→  // The financial account primarily affected (receiving account for income,
   733→  // paying account for expense, source for transfer). For multi-account
   734→  // journals (transfers), the full debit/credit breakdown is in JournalEntry.
   735→  financialAccountId String?
   736→  financialAccount   FinancialAccount? @relation(fields: [financialAccountId], references: [id], onDelete: Restrict)
   737→
   738→  // The primary ledger category (income/expense category). Full breakdown
   739→  // is in JournalEntry.
   740→  ledgerAccountId String?
   741→  ledgerAccount   LedgerAccount? @relation(fields: [ledgerAccountId], references: [id], onDelete: Restrict)
   742→
   743→  // Optional relations — wired to Phase 5/6 entities when they land.
   744→  departmentId   String?
   745→  department     Department? @relation(fields: [departmentId], references: [id], onDelete: SetNull)
   746→
   747→  // Party reference (customer/supplier) — nullable stub from Phase 2.
   748→  // `partyType` discriminates; `partyRef` holds the entity ID as a string.
   749→  // Phase 4 adds proper FK relations (customerId/supplierId) below.
   750→  partyType String? // customer | supplier | null
   751→  partyRef  String? // future FK to Customer/Supplier

   756→  customer   Customer? @relation("JournalCustomer", fields: [customerId], references: [id], onDelete: SetNull)
   757→  supplierId String?
   758→  supplier   Supplier? @relation("JournalSupplier", fields: [supplierId], references: [id], onDelete: SetNull)
   759→
   760→  // Project reference — legacy string stub from Phase 2.

   762→  projectRef String?
   763→
   764→  // Phase 5: proper Project FK relation (nullable, alongside legacy stub).
   765→  projectId String?
   766→  project   Project? @relation("JournalProject", fields: [projectId], references: [id], onDelete: SetNull)
   767→
   768→  paymentMethod String? // cash | bank_transfer | mobile_money | card | cheque | other
   769→  externalRef  String? // bank slip / cheque / momo transaction id
   770→
   771→  // Reversal linkage. `reversesId` is the FK on the reversal journal pointing
   772→  // to the original. `reverses` is the relation (holds fields). `reversedBy`
   773→  // is the inverse relation on the original journal (no fields).

   776→  reversedBy     Journal?  @relation("JournalReversal")

   781→  amount      Decimal
   782→  currency    String  @default("GHS")

   786→  postedAt    DateTime?
   787→  voidedAt    DateTime?

   789→  voidedBy    User?   @relation("JournalVoidedBy", fields: [voidedById], references: [id], onDelete: SetNull)

   796→  @@index([status])

   808→// A single debit OR credit line within a Journal. Exactly one of
   809→// (debit, credit) is non-zero per entry. The journal is balanced when
   810→// Σ(debit) = Σ(credit) across all its entries.
   811→//
   812→// `financialAccountId` is OPTIONAL: only entries that move money in/out of
   813→// a cash/bank/momo account reference a FinancialAccount. The counter-side
   814→// (e.g. an income category credit, or an expense category debit) references
   815→// only a LedgerAccount. This mirrors real accounting where income/expense
   816→// categories are not "accounts you hold money in".
   817→model JournalEntry {
   818→  id                String   @id @default(cuid())
   819→  journalId         String
   820→  journal           Journal  @relation(fields: [journalId], references: [id], onDelete: Cascade)
   821→
   822→  financialAccountId String?
   823→  financialAccount   FinancialAccount? @relation(fields: [financialAccountId], references: [id], onDelete: Restrict)
   824→
   825→  ledgerAccountId   String?
   826→  ledgerAccount     LedgerAccount? @relation(fields: [ledgerAccountId], references: [id], onDelete: Restrict)
   827→
   828→  // Exactly one of debit/credit is non-zero (the other is 0). Both Decimal.
   829→  debit  Decimal @default(0)
   830→  credit Decimal @default(0)
   831→  currency String @default("GHS")
   832→
   833→  description String?
   834→  createdAt   DateTime @default(now())
   835→
   836→  @@index([journalId])
   837→  @@index([financialAccountId])
   838→  @@index([ledgerAccountId])
   839→}
